rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /reviews/{reviewId} {
      allow read: if true;
      allow create: if true;
      allow update, delete: if false;
    }
    match /{document=**} {
      allow read, write: if false;
    }
  }
}

NOTA: Estas reglas son deliberadamente simples para comprobar la conexión. Cuando confirmemos que funciona, podemos endurecer la validación sin romper el formulario.
